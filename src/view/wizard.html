Мастер форм (пошаговая форма)

Три шага, у каждого — своё условие, разрешающее переход дальше: на первом должно быть заполнено имя, на втором — введён похожий на email адрес. Проверка нужна сразу в двух местах — чтобы отключить кнопку «Далее» и чтобы скрыть её на последнем шаге, — поэтому она вынесена в метод canGoNext(), а обе кнопки навигации оформлены наборами back/next и подключены через v-bind. 

document.addEventListener('youla:init', () => {
  Youla.data('wizard', () => ({
    step: 1,
    name: '',
    email: '',
    isStep(n) {
      return this.step === n;
    },
    canGoNext() {
      if (this.step === 1) return this.name.trim().length > 0;
      if (this.step === 2) return /.+@.+\..+/.test(this.email);
      return false;
    },
    goNext() {
      if (this.canGoNext()) {
        this.step++;
      }
    },
    goBack() {
      this.step--;
    },
    back: {
      '@click'() {
        this.goBack();
      },
      ':disabled'() {
        return this.step <= 1;
      },
    },
    next: {
      '@click'() {
        this.goNext();
      },
      ':disabled'() {
        return !this.canGoNext();
      },
      'v-show'() {
        return this.step < 3;
      },
    },
  }));
});

<div v-data="wizard">
  <div v-show="isStep(1)">
    <label>Ваше имя</label>
    <input v-prop="name" placeholder="Иван">
  </div>

  <div v-show="isStep(2)">
    <label>Email</label>
    <input v-prop="email" placeholder="ivan@example.com">
  </div>

  <div v-show="isStep(3)">
    Готово! <span v-text="name"></span>, мы отправим подтверждение на <span v-text="email"></span>.
  </div>

  <div>
    <button type="button" v-bind="back">Назад</button>
    <button type="button" v-bind="next">Далее</button>
  </div>
</div>

Каждый шаг (v-show="isStep(n)") остался обычным атрибутом — на этих элементах всего одна динамическая привязка, оборачивать её в v-bind только ради формы не нужно. А вот у обеих кнопок навигации по два-три атрибута сразу, и вынесение их в back/next избавляет разметку от повторяющихся :disabled и v-show.
