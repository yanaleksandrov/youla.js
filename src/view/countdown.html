Обратный отсчёт до даты

Таймер запускается один раз и сам пересчитывает дни/часы/минуты/секунды каждую секунду через setInterval. Запустить его нужно сразу при появлении компонента на странице — для этого подходит специальное событие @load, которое Youla.js вызывает сразу после инициализации элемента (подробнее — на странице События), а не обычный @click. 

document.addEventListener('youla:init', () => {
  Youla.data('countdown', () => ({
    end: new Date(Date.now() + 65 * 1000).toISOString(),
    day:  '00',
    hour: '00',
    min:  '00',
    sec:  '00',
    start() {
      let end = new Date(this.end).valueOf();

      this.timer = setInterval(() => {
        let diff = Math.max(0, Math.round((end - Date.now()) / 1000));

        this.day  = String(Math.floor(diff / 86400)).padStart(2, '0');
        this.hour = String(Math.floor((diff / 3600) % 24)).padStart(2, '0');
        this.min  = String(Math.floor((diff / 60) % 60)).padStart(2, '0');
        this.sec  = String(diff % 60).padStart(2, '0');

        if (diff <= 0) {
          clearInterval(this.timer);
        }
      }, 1000);
    },
  }));
});

<div v-data="countdown" @load="start()">
  <span v-text="day"></span> дн.
  <span v-text="hour"></span> ч.
  <span v-text="min"></span> мин.
  <span v-text="sec"></span> сек.
</div>
дн. ч. мин. сек.

В примере обратный отсчёт настроен всего на 65 секунд, чтобы можно было увидеть его завершение, не дожидаясь реальной даты, — в боевом коде end обычно приходит с сервера или вычисляется от настоящей даты события.
