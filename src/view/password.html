Пароль: показать и оценить надёжность

Три места, где копится по несколько атрибутов сразу: поле ввода совмещает v-prop и переключение :type, кнопка — клик и подпись, полоска надёжности — класс уровня и ширину. Каждая пара оформлена отдельным набором и подключена через v-bind, поэтому в разметке не осталось ни одного элемента с двумя атрибутами одновременно. 

document.addEventListener('youla:init', () => {
  Youla.data('password', () => ({
    visible: false,
    value: '',
    labels: ['Слишком слабый', 'Слабый', 'Средний', 'Хороший', 'Отличный'],
    toggle() {
      this.visible = !this.visible;
    },
    score() {
      let value = this.value, points = 0;

      if (value.length >= 8) points++;
      if (/[A-ZА-Я]/.test(value)) points++;
      if (/\d/.test(value)) points++;
      if (/[^A-Za-zА-Яа-я0-9]/.test(value)) points++;

      return points;
    },
    label() {
      return this.labels[this.score()];
    },
    field: {
      ':type': "visible ? 'text' : 'password'",
      'v-prop': 'value',
    },
    toggler: {
      '@click'() {
        this.toggle();
      },
      'v-text'() {
        return this.visible ? 'Скрыть' : 'Показать';
      },
    },
    bar: {
      ':class': '`level-${score()}`',
      ':style': "{ width: `${score() * 25}%` }",
    },
  }));
});

<div v-data="password">
  <input v-bind="field" placeholder="Придумайте пароль">
  <button type="button" v-bind="toggler"></button>

  <div class="password-meter">
    <div class="password-meter__bar" v-bind="bar"></div>
  </div>
  <div v-text="value && label()"></div>
</div>

field и bar — обычные свойства со строками-выражениями (они пересчитываются реактивно точно так же, как если бы были написаны в разметке напрямую), а toggler — пара методов, потому что кнопке нужен доступ к this для вызова toggle().
